No primeiro uso, crie o usuário no Supabase Auth e depois ajuste o campo role em profiles para admin ou user.

SQL e policies

Cria auth, profiles, front-end admin e área restrita do usuário.

create extension if not exists pgcrypto; create table if not exists public.profiles ( id uuid primary key references auth.users(id) on delete cascade, nome text, role text not null default 'user', created_at timestamp with time zone default now() ); create table if not exists public.boloes ( id uuid primary key default gen_random_uuid(), titulo text not null, loteria text not null, valor_cota text not null, data_sorteio date not null, data_pagamento date not null, pix text not null, slug text unique not null, descricao text, created_at timestamp with time zone default now() ); create table if not exists public.participantes ( id uuid primary key default gen_random_uuid(), bolao_id uuid references public.boloes(id) on delete cascade, user_id uuid references auth.users(id) on delete set null, nome text not null, contato text, status text default 'pendente', quantidade_cotas integer default 1, created_at timestamp with time zone default now() ); create table if not exists public.jogos ( id uuid primary key default gen_random_uuid(), bolao_id uuid references public.boloes(id) on delete cascade, concurso text not null, dezenas text not null, valor text, observacoes text, created_at timestamp with time zone default now() ); alter table public.participantes add column if not exists user_id uuid references auth.users(id) on delete set null; alter table public.participantes add column if not exists quantidade_cotas integer default 1; alter table public.profiles enable row level security; alter table public.boloes enable row level security; alter table public.participantes enable row level security; alter table public.jogos enable row level security; create or replace function public.is_admin() returns boolean language sql stable as $$ select exists ( select 1 from public.profiles where id = auth.uid() and role = 'admin' ); $$; create or replace function public.handle_new_user() returns trigger language plpgsql security definer as $$ begin insert into public.profiles (id, nome, role) values (new.id, coalesce(new.raw_user_meta_data->>'nome', split_part(new.email, '@', 1)), 'user') on conflict (id) do nothing; return new; end; $$; drop trigger if exists on_auth_user_created on auth.users; create trigger on_auth_user_created after insert on auth.users for each row execute procedure public.handle_new_user(); -- profiles drop policy if exists "profiles_select_own_or_admin" on public.profiles; drop policy if exists "profiles_update_own_or_admin" on public.profiles; create policy "profiles_select_own_or_admin" on public.profiles for select to authenticated using (id = auth.uid() or public.is_admin()); create policy "profiles_update_own_or_admin" on public.profiles for update to authenticated using (id = auth.uid() or public.is_admin()) with check (id = auth.uid() or public.is_admin()); -- boloes drop policy if exists "boloes_select_all_authenticated" on public.boloes; drop policy if exists "boloes_admin_insert" on public.boloes; drop policy if exists "boloes_admin_update" on public.boloes; drop policy if exists "boloes_admin_delete" on public.boloes; create policy "boloes_select_all_authenticated" on public.boloes for select to authenticated using (true); create policy "boloes_admin_insert" on public.boloes for insert to authenticated with check (public.is_admin()); create policy "boloes_admin_update" on public.boloes for update to authenticated using (public.is_admin()) with check (public.is_admin()); create policy "boloes_admin_delete" on public.boloes for delete to authenticated using (public.is_admin()); -- participantes drop policy if exists "participantes_select_admin_or_own" on public.participantes; drop policy if exists "participantes_insert_admin_or_self" on public.participantes; drop policy if exists "participantes_update_admin_or_own" on public.participantes; drop policy if exists "participantes_delete_admin_only" on public.participantes; create policy "participantes_select_admin_or_own" on public.participantes for select to authenticated using (public.is_admin() or user_id = auth.uid()); create policy "participantes_insert_admin_or_self" on public.participantes for insert to authenticated with check (public.is_admin() or user_id = auth.uid()); create policy "participantes_update_admin_or_own" on public.participantes for update to authenticated using (public.is_admin() or user_id = auth.uid()) with check (public.is_admin() or user_id = auth.uid()); create policy "participantes_delete_admin_only" on public.participantes for delete to authenticated using (public.is_admin()); -- jogos drop policy if exists "jogos_select_all_authenticated" on public.jogos; drop policy if exists "jogos_admin_insert" on public.jogos; drop policy if exists "jogos_admin_update" on public.jogos; drop policy if exists "jogos_admin_delete" on public.jogos; create policy "jogos_select_all_authenticated" on public.jogos for select to authenticated using (true); create policy "jogos_admin_insert" on public.jogos for insert to authenticated with check (public.is_admin()); create policy "jogos_admin_update" on public.jogos for update to authenticated using (public.is_admin()) with check (public.is_admin()); create policy "jogos_admin_delete" on public.jogos for delete to authenticated using (public.is_admin());
